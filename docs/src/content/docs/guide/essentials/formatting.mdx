import { plainDate, zonedDateTime } from "../../../../components/temporal";
import { period } from "@aleksejdix/minuta";
import { formatPeriod } from "@aleksejdix/minuta/format";
import Timeline from "../../../../components/diagrams/Timeline.astro";

export const moment = zonedDateTime("2026-03-18T14:30[Europe/Zurich]");
export const labelled = (units) =>
  units.map((unit) => ({
    label: `${unit}: ${formatPeriod(period(moment, unit), "en-US")}`,
    periods: [period(moment, unit)],
    tone: "accent",
  }));
export const weeks = [
  period(plainDate("2025-12-31"), "week"),
  period(plainDate("2026-03-02"), "week"),
  period(plainDate("2026-03-31"), "week"),
];

A period is data. Sooner or later you show it to a person: "March 2026" above
a calendar, "Wednesday, March 18" on a day view, "Mar 2 – 8, 2026" on a week.
`@aleksejdix/minuta/format` writes these labels for you, in any language.

```ts
import { formatPeriod, formatRange } from "@aleksejdix/minuta/format";
```

Both functions use the `Intl.DateTimeFormat` built into browsers and Node, so
minuta ships no language data. They only read `start`, `end` and `unit`, so
they work with periods from any units.

All examples run in Europe/Zurich.

## A title for the calendar header

Pass the period and a locale to
[`formatPeriod`](../../../api/minuta-format/format-period/):

```ts
import { next, period } from "@aleksejdix/minuta";
import { formatPeriod } from "@aleksejdix/minuta/format";

const shown = period(Temporal.PlainDate.from("2026-03-18"), "month");

formatPeriod(shown, "en-US"); //       → "March 2026"
formatPeriod(next(shown), "en-US"); // → "April 2026"
```

Together with the ‹ › buttons from [Navigating time](../navigating-time/),
that is a working calendar header: step `shown`, then format it again.

## The unit picks the format

You don't choose a pattern. `formatPeriod` looks at the period's unit and
shows what makes sense for it. For Wed Mar 18 2026, 14:30, in `"en-US"`:

| Unit      | Label             |
| --------- | ----------------- |
| `year`    | 2026              |
| `quarter` | Jan 2026          |
| `month`   | March 2026        |
| `week`    | Mar 16 – 22, 2026 |
| `day`     | March 18, 2026    |
| `hour`    | Mar 18, 2:00 PM   |
| `minute`  | Mar 18, 2:30 PM   |

So the same line of code labels a day view, a week view and a month view.

ones:

:::caution
A quarter is labelled by its **first month**: Q2 2026 shows as "Apr 2026",
not as a quarter. To show all three months, use `formatRange`:

```ts
const q2 = period(Temporal.PlainDate.from("2026-05-01"), "quarter");

formatPeriod(q2, "en-US"); // → "Apr 2026"
formatRange(q2, "en-US"); //  → "Apr – Jun 2026"
```

:::

## Day labels

A day period shows the full date. Pick other fields with `Intl` options.
Options that pick fields (`weekday`, `day`, `month`, `year` …) **replace** the
defaults, so name every field you want:

```ts
const day = period(Temporal.PlainDate.from("2026-03-18"), "day");

formatPeriod(day, "en-US"); // → "March 18, 2026"
formatPeriod(day, "en-US", { weekday: "long" }); // → "Wednesday"
formatPeriod(day, "en-US", { weekday: "long", month: "long", day: "numeric" });
// → "Wednesday, March 18"
```

That is how you get the short labels of a calendar grid, still in the
reader's language:

```ts
const cell = { day: "numeric" } as const;

formatPeriod(day, "en-US", cell); // → "18"
formatPeriod(day, "ja-JP", cell); // → "18日"
```

The same works for the weekday names above the grid:

```ts
const weekday = { weekday: "short" } as const;

formatPeriod(day, "en-US", weekday); // → "Wed"
formatPeriod(day, "de-CH", weekday); // → "Mi"
```

## Weeks and other ranges

A week is shown as a range from its first to its last day (the day before
`end`). Shared parts, such as the month and year, are written only once:

```ts
const week = period(Temporal.PlainDate.from("2026-03-02"), "week");

formatPeriod(week, "en-US"); // → "Mar 2 – 8, 2026"
formatPeriod(period(Temporal.PlainDate.from("2026-03-31"), "week"), "en-US");
// → "Mar 30 – Apr 5, 2026"
formatPeriod(period(Temporal.PlainDate.from("2025-12-31"), "week"), "en-US");
// → "Dec 29, 2025 – Jan 4, 2026"
```

The three weeks on one axis. The more the start and the end have in common,
the shorter the label:

<Timeline
  caption="A week across two years, a week inside one month and a week across two months, each labelled with formatPeriod in en-US."
  tracks={weeks.map((week) => ({
    label: formatPeriod(week, "en-US"),
    periods: [week],
    tone: "accent",
  }))}
/>

Custom periods from `range` are shown as a range too:

```ts
import { range } from "@aleksejdix/minuta";

formatPeriod(
  range(
    Temporal.PlainDate.from("2026-03-30"),
    Temporal.PlainDate.from("2026-04-06")
  ),
  "en-US"
);
// → "Mar 30 – Apr 5, 2026", the last day before the end
```

[`formatRange`](../../../api/minuta-format/format-range/) always shows a
period as "start – end", whatever its unit. When start and end look the same
in the shown fields, it collapses to one:

```ts
import { formatRange } from "@aleksejdix/minuta/format";

formatRange(
  period(
    Temporal.ZonedDateTime.from("2026-03-18T14:00[Europe/Zurich]"),
    "hour"
  ),
  "en-US"
);
// → "Mar 18, 2:00 – 3:00 PM"

formatRange(period(Temporal.PlainDate.from("2026-03-18"), "day"), "en-US");
// → "March 18, 2026"
```

A time period shows its `end` as it is: the hour from 2:00 ends at 3:00. A
date period shows its last day, so a day collapses to one date.

## Times in a range

A custom range shows only dates by default, so the times of a meeting are not
shown. Name the fields you want, dates and times:

```ts
const meeting = range(
  Temporal.ZonedDateTime.from("2026-03-18T09:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-18T10:30[Europe/Zurich]")
);
const dateAndTime = {
  day: "numeric",
  month: "short",
  year: "numeric",
  hour: "numeric",
  minute: "2-digit",
} as const;

formatRange(meeting, "en-US"); // → "Mar 18, 2026"
formatRange(meeting, "en-US", dateAndTime);
// → "Mar 18, 2026, 9:00 – 10:30 AM"
formatRange(meeting, "en-US", { hour: "numeric", minute: "2-digit" });
// → "9:00 – 10:30 AM"
```

## Locales

The locale is a BCP 47 tag. The order of the parts, the month names and the
punctuation all follow it:

```ts
const march = period(Temporal.PlainDate.from("2026-03-18"), "month");

formatPeriod(march, "en-US"); // → "March 2026"
formatPeriod(march, "de-CH"); // → "März 2026"
formatPeriod(march, "fr-FR"); // → "mars 2026"
formatPeriod(march, "ja-JP"); // → "2026年3月"

formatPeriod(week, "de-CH"); // → "2.–8. März 2026"
formatPeriod(week, "en-GB"); // → "2 – 8 Mar 2026"
```

The locale can also ask for another calendar system with `-u-ca-`. Only the
label changes; the period is still the same month:

```ts
formatPeriod(march, "en-US-u-ca-buddhist"); // → "March 2569 BE"
```

## Intl options

Any [`Intl.DateTimeFormat` option](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/Intl/DateTimeFormat/DateTimeFormat#options)
works. A few that come up often:

```ts
formatPeriod(march, "en-US", { month: "short", year: "numeric" }); // → "Mar 2026"
formatPeriod(day, "en-US", {
  month: "2-digit",
  day: "2-digit",
  year: "numeric",
});
// → "03/18/2026"
formatPeriod(day, "en-US", { dateStyle: "full" }); // → "Wednesday, March 18, 2026"
formatPeriod(
  period(
    Temporal.ZonedDateTime.from("2026-03-18T14:00[Europe/Zurich]"),
    "hour"
  ),
  "en-US",
  { hourCycle: "h23" }
);
// → "Mar 18, 14:00"
```

Options that do not pick fields, such as `hourCycle`, are added to the
defaults instead of replacing them.

An invalid locale tag throws a `RangeError`, also from `Intl.DateTimeFormat`.

## Time zones

A date period has no zone, so its label is the same everywhere, and
`timeZone` does not change it. A time period is shown in the zone of its
`ZonedDateTime`s, not the computer's:

```ts
import { createUnits, withUnits } from "@aleksejdix/minuta/core";
import { formatPeriod } from "@aleksejdix/minuta/format";

const tokyo = withUnits(createUnits({ timeZone: "Asia/Tokyo" }));
const tokyoHour = tokyo.period(
  Temporal.Instant.from("2026-03-20T20:00:00Z"),
  "hour"
);

formatPeriod(tokyoHour, "en-US"); // → "Mar 21, 5:00 AM", on any computer
```

It is already 21 March in Tokyo, but still 20 March in Zurich. The label
follows the period's zone.

To show a time period in another zone, pass `timeZone`, for example the same
hour for a colleague in New York:

```ts
formatPeriod(
  period(
    Temporal.ZonedDateTime.from("2026-03-18T14:00[Europe/Zurich]"),
    "hour"
  ),
  "en-US",
  { timeZone: "America/New_York" }
);
// → "Mar 18, 9:00 AM"
```

See [Time zones](../../../concepts/time-zones/) for the full picture.
