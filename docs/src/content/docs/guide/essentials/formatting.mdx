import { plainDate, zonedDateTime } from "../../../../components/temporal";
import { period, toLocaleString } from "@aleksejdix/minuta";
import Timeline from "../../../../components/diagrams/Timeline.astro";

export const moment = zonedDateTime("2026-03-18T14:30[Europe/Zurich]");
export const labelled = (units) =>
  units.map((unit) => ({
    label: `${unit}: ${toLocaleString(period(moment, unit), "en-US")}`,
    periods: [period(moment, unit)],
    tone: "accent",
  }));
export const weeks = [
  period(plainDate("2025-12-31"), "week"),
  period(plainDate("2026-03-02"), "week"),
  period(plainDate("2026-03-31"), "week"),
];

A period is data. Sooner or later you show it to a person: "March 2026" above
a calendar, "Wednesday, March 18" on a day view, "Mar 2 – 8, 2026" on a week.
`toLocaleString` writes these labels for you, in any language, like
Temporal's `toLocaleString()`.

```ts
import { toLocaleString } from "@aleksejdix/minuta";
```

It uses the `Intl.DateTimeFormat` built into browsers and Node, so minuta
ships no language data. It only reads `start`, `end` and `unit`, so it works
with periods from any units.

All examples run in Europe/Zurich.

## A title for the calendar header

Pass the period and a locale to
[`toLocaleString`](../../../api/minuta/to-locale-string/):

```ts
import { next, period, toLocaleString } from "@aleksejdix/minuta";

const shown = period(Temporal.PlainDate.from("2026-03-18"), "month");

toLocaleString(shown, "en-US"); //       → "March 2026"
toLocaleString(next(shown), "en-US"); // → "April 2026"
```

Together with the ‹ › buttons from [Navigating time](../navigating-time/),
that is a working calendar header: step `shown`, then format it again.

## The unit picks the format

You don't choose a pattern. `toLocaleString` looks at the period's unit and
shows what makes sense for it. For Wed Mar 18 2026, 14:30, in `"en-US"`:

| Unit      | Label             |
| --------- | ----------------- |
| `year`    | 2026              |
| `quarter` | Jan 2026          |
| `month`   | March 2026        |
| `week`    | Mar 16 – 22, 2026 |
| `day`     | March 18, 2026    |
| `hour`    | Mar 18, 2:00 PM   |
| `minute`  | Mar 18, 2:30 PM   |

So the same line of code labels a day view, a week view and a month view.

ones:

Then the small ones, on the scale of one week:

:::caution
A quarter is labelled by its **first month**: Q2 2026 shows as "Apr 2026",
not as a quarter. To show all three months, format it as a range, with the
fields you want:

```ts
import { period, range, toLocaleString } from "@aleksejdix/minuta";

const q2 = period(Temporal.PlainDate.from("2026-05-01"), "quarter");

toLocaleString(q2, "en-US"); // → "Apr 2026"
toLocaleString(range(q2.start, q2.end), "en-US"); // → "Apr 1 – Jun 30, 2026"
toLocaleString(range(q2.start, q2.end), "en-US", {
  month: "short",
  year: "numeric",
}); // → "Apr – Jun 2026"
```

:::

## Day labels

A day period shows the full date. Pick other fields with `Intl` options.
Options that pick fields (`weekday`, `day`, `month`, `year` …) **replace** the
defaults, so name every field you want:

```ts
const day = period(Temporal.PlainDate.from("2026-03-18"), "day");

toLocaleString(day, "en-US"); // → "March 18, 2026"
toLocaleString(day, "en-US", { weekday: "long" }); // → "Wednesday"
toLocaleString(day, "en-US", {
  weekday: "long",
  month: "long",
  day: "numeric",
});
// → "Wednesday, March 18"
```

That is how you get the short labels of a calendar grid, still in the
reader's language:

```ts
const cell = { day: "numeric" } as const;

toLocaleString(day, "en-US", cell); // → "18"
toLocaleString(day, "ja-JP", cell); // → "18日"
```

The same works for the weekday names above the grid:

```ts
const weekday = { weekday: "short" } as const;

toLocaleString(day, "en-US", weekday); // → "Wed"
toLocaleString(day, "de-CH", weekday); // → "Mi"
```

## Weeks and other ranges

A week is shown as a range from its first to its last day (the day before
`end`). Shared parts, such as the month and year, are written only once:

```ts
const week = period(Temporal.PlainDate.from("2026-03-02"), "week");

toLocaleString(week, "en-US"); // → "Mar 2 – 8, 2026"
toLocaleString(period(Temporal.PlainDate.from("2026-03-31"), "week"), "en-US");
// → "Mar 30 – Apr 5, 2026"
toLocaleString(period(Temporal.PlainDate.from("2025-12-31"), "week"), "en-US");
// → "Dec 29, 2025 – Jan 4, 2026"
```

The three weeks on one axis. The more the start and the end have in common,
the shorter the label:

<Timeline
  caption="A week across two years, a week inside one month and a week across two months, each labelled with toLocaleString in en-US."
  tracks={weeks.map((week) => ({
    label: toLocaleString(week, "en-US"),
    periods: [week],
    tone: "accent",
  }))}
/>

Custom periods from `range` are shown as a range too:

```ts
import { period, range } from "@aleksejdix/minuta";

toLocaleString(
  range(
    Temporal.PlainDate.from("2026-03-30"),
    Temporal.PlainDate.from("2026-04-06")
  ),
  "en-US"
);
// → "Mar 30 – Apr 5, 2026", the last day before the end
```

To show any period as "start – end", whatever its unit, format a `range`
with the same start and end. When start and end look the same in the shown
fields, it collapses to one:

```ts
const hour = period(
  Temporal.ZonedDateTime.from("2026-03-18T14:00[Europe/Zurich]"),
  "hour"
);
toLocaleString(range(hour.start, hour.end), "en-US");
// → "Mar 18, 2:00 – 3:00 PM"

const day = period(Temporal.PlainDate.from("2026-03-18"), "day");
toLocaleString(range(day.start, day.end), "en-US");
// → "Mar 18, 2026"
```

A time period shows its `end` as it is: the hour from 2:00 ends at 3:00. A
date period shows its last day, so a day collapses to one date.

## Times in a range

A custom range of moments shows its day and times by default. Name the
fields you want to show more or less:

```ts
const meeting = range(
  Temporal.ZonedDateTime.from("2026-03-18T09:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-18T10:30[Europe/Zurich]")
);
const dateAndTime = {
  day: "numeric",
  month: "short",
  year: "numeric",
  hour: "numeric",
  minute: "2-digit",
} as const;

toLocaleString(meeting, "en-US"); // → "Mar 18, 9:00 – 10:30 AM"
toLocaleString(meeting, "en-US", dateAndTime);
// → "Mar 18, 2026, 9:00 – 10:30 AM"
toLocaleString(meeting, "en-US", { hour: "numeric", minute: "2-digit" });
// → "9:00 – 10:30 AM"
```

## Locales

The locale is a BCP 47 tag. The order of the parts, the month names and the
punctuation all follow it:

```ts
const march = period(Temporal.PlainDate.from("2026-03-18"), "month");

toLocaleString(march, "en-US"); // → "March 2026"
toLocaleString(march, "de-CH"); // → "März 2026"
toLocaleString(march, "fr-FR"); // → "mars 2026"
toLocaleString(march, "ja-JP"); // → "2026年3月"

toLocaleString(week, "de-CH"); // → "2.–8. März 2026"
toLocaleString(week, "en-GB"); // → "2 – 8 Mar 2026"
```

The locale can also ask for another calendar system with `-u-ca-`. Only the
label changes; the period is still the same month:

```ts
toLocaleString(march, "en-US-u-ca-buddhist"); // → "March 2569 BE"
```

## Intl options

Any [`Intl.DateTimeFormat` option](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/Intl/DateTimeFormat/DateTimeFormat#options)
works. A few that come up often:

```ts
toLocaleString(march, "en-US", { month: "short", year: "numeric" }); // → "Mar 2026"
toLocaleString(day, "en-US", {
  month: "2-digit",
  day: "2-digit",
  year: "numeric",
});
// → "03/18/2026"
toLocaleString(day, "en-US", { dateStyle: "full" }); // → "Wednesday, March 18, 2026"
toLocaleString(
  period(
    Temporal.ZonedDateTime.from("2026-03-18T14:00[Europe/Zurich]"),
    "hour"
  ),
  "en-US",
  { hourCycle: "h23" }
);
// → "Mar 18, 14:00"
```

Options that do not pick fields, such as `hourCycle`, are added to the
defaults instead of replacing them.

An invalid locale tag throws a `RangeError`, also from `Intl.DateTimeFormat`.

## Time zones

A date period has no zone, so its label is the same everywhere, and
`timeZone` does not change it. A time period is shown in the zone of its
`ZonedDateTime`s, not the computer's:

```ts
import { period, toLocaleString } from "@aleksejdix/minuta";

const tokyoHour = period(
  Temporal.Instant.from("2026-03-20T20:00:00Z").toZonedDateTimeISO(
    "Asia/Tokyo"
  ),
  "hour"
);

toLocaleString(tokyoHour, "en-US"); // → "Mar 21, 5:00 AM", on any computer
```

It is already 21 March in Tokyo, but still 20 March in Zurich. The label
follows the period's zone.

To show a time period in another zone, pass `timeZone`, for example the same
hour for a colleague in New York:

```ts
toLocaleString(
  period(
    Temporal.ZonedDateTime.from("2026-03-18T14:00[Europe/Zurich]"),
    "hour"
  ),
  "en-US",
  { timeZone: "America/New_York" }
);
// → "Mar 18, 9:00 AM"
```

See [Time zones](../../../concepts/time-zones/) for the full picture.

## Text for storage and URLs

`toLocaleString` writes text for people. To store a period, or put it in a
URL or an API, write it as an ISO 8601 interval: `start/end`. Temporal
already writes and reads each edge, so this takes no minuta function:

```ts
import { period, range } from "@aleksejdix/minuta";

const toText = (period) => `${period.start}/${period.end}`;

// The `/` between the edges, not the one in a zone such as [Europe/Zurich]
const BETWEEN = /\/(?![^[]*\])/u;

function fromText(text) {
  const [start, end] = text.split(BETWEEN);
  const read = start.includes("T")
    ? Temporal.ZonedDateTime.from
    : Temporal.PlainDate.from;
  return range(read(start), read(end));
}

toText(period(Temporal.PlainDate.from("2026-03-18"), "month"));
// → "2026-03-01/2026-04-01"
toText(
  period(Temporal.ZonedDateTime.from("2026-03-18T14:30[Europe/Zurich]"), "hour")
);
// → "2026-03-18T14:00:00+01:00[Europe/Zurich]/2026-03-18T15:00:00+01:00[Europe/Zurich]"
```

Three things to know:

- Do not split at every `/`: time zone names have one too.
- A moment needs its time zone in brackets, which `ZonedDateTime` writes; a
  calendar annotation such as `[u-ca=hebrew]` keeps a date in its calendar.
- The text names a range, not a unit, so `fromText` gives a `"custom"`
  period. [`merge`](../../../api/minuta/merge/) gives it back the unit it
  lines up with: `merge([fromText("2026-03-01/2026-04-01")], "month")`.

ISO 8601 also allows a duration for one edge, `2026-03-01/P1M`.
`Temporal.Duration.from` reads it; add it to the start with `start.add`.
