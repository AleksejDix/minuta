  zonedDateTime("2026-03-22T23:00[Europe/Zurich]"),
  zonedDateTime("2026-03-23T01:00[Europe/Zurich]")

export const trip = period(plainDate("2026-03-18"), plainDate("2026-03-25"));
export const wholeTrip = period(
  day(plainDate("2026-03-18")).start,
  day(plainDate("2026-03-25")).end
);

Once you have periods, you ask questions about them. Is this event today?
Does it fall in the week on screen? Do two bookings clash? How many days does
this month have? Each question has one function.

| Question                                  | Function                                               |
| ----------------------------------------- | ------------------------------------------------------ |
| Is this date or period inside that one?   | [`contains`](../../../api/minuta/contains/)            |
| Are these two in the same day, week, …?   | [`equals`](../../../api/minuta/equals/) of their units |
| Do these two share any time?              | [`overlaps`](../../../api/minuta/overlaps/)            |
| Are these two the very same period?       | [`equals`](../../../api/minuta/equals/)                |
| Which comes first?                        | [`compare`](../../../api/minuta/compare/)              |
| How many whole days, hours, … fit inside? | [`count`](../../../api/minuta/count/)                  |
| How long is it, in exact time?            | `period.start.until(period.end)`                       |

All examples run in Europe/Zurich, with weeks from Monday to Sunday.

## Is this today?

Make the period for today, then ask whether it contains the date or moment:

```ts
import { contains, day } from "@aleksejdix/minuta";

const today = day(Temporal.PlainDate.from("2026-03-18"));

contains(today, Temporal.ZonedDateTime.from("2026-03-18T09:00[Europe/Zurich]")); // → true
contains(today, Temporal.PlainDate.from("2026-03-19")); //                          → false, that is tomorrow
```

A period includes its `start` but not its `end`. The `end` of today is
already tomorrow:

```ts
today.end; //                        → 2026-03-19
contains(today, today.end); //       → false
contains(today, next(today).start); // → false, the same date
```

## Same day, same week, same month

To compare two periods at the level you choose, take the unit around each
start and ask [`equals`](../../../api/minuta/equals/). The periods can have
any units of their own:

```ts
import { day, equals, hour, month, week } from "@aleksejdix/minuta";

const today = day(Temporal.PlainDate.from("2026-03-18")); // a Wednesday
const meeting = hour(
  Temporal.ZonedDateTime.from("2026-03-18T09:00[Europe/Zurich]")
);

equals(day(meeting.start), today); // → true

equals(week(today.start), week(Temporal.PlainDate.from("2026-03-17"))); // → true, Tuesday
equals(week(today.start), week(Temporal.PlainDate.from("2026-03-23"))); // → false, next Monday

equals(month(today.start), month(Temporal.PlainDate.from("2025-03-18"))); // → false, March 2025
```

For a unit held in a variable, use a bound object:
`equals(time[unit](a.start), time[unit](b.start))`, with
`time = withUnits(createUnits())`.

That last line shows that "same month" means the same month _of the same
year_, not just the same month name.

This is how a calendar highlights today's cell: `equals(cell, today)`, as
both are day periods.

:::caution
This compares the units around the **starts** only. The week Mon Mar 30 – Sun
Apr 5 counts as March, even though most of it is in April:

```ts
const weekOfMar31 = week(Temporal.PlainDate.from("2026-03-31"));

equals(month(weekOfMar31.start), month(Temporal.PlainDate.from("2026-03-01"))); // → true
equals(month(weekOfMar31.start), month(Temporal.PlainDate.from("2026-04-01"))); // → false
```

To ask whether the whole week lies in a month, use `contains`.
:::

## The very same period, and in order

[`equals`](../../../api/minuta/equals/) asks whether two periods are
identical: the same unit, start and end, like Temporal's `equals`.
[`compare`](../../../api/minuta/compare/) orders periods by start, then by
end, like Temporal's `compare`, so it sorts a list:

```ts
import { compare, equals, month, next, period } from "@aleksejdix/minuta";

const march = month(Temporal.PlainDate.from("2026-03-18"));

equals(march, month(Temporal.PlainDate.from("2026-03-01"))); // → true
equals(march, period(march.start, march.end)); // → false, a month and a custom period

[next(march), march].sort(compare); // → [march, april]
```

## Does this event fall in the week?

`contains` also takes a period. It is `true` only when the whole period lies
inside:

```ts
import { contains, period, week } from "@aleksejdix/minuta";

const thisWeek = week(Temporal.PlainDate.from("2026-03-18"));
// → { start: 2026-03-16, end: 2026-03-23, unit: "week" }

const meeting = period(
  Temporal.ZonedDateTime.from("2026-03-20T09:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-20T10:00[Europe/Zurich]")
);
contains(thisWeek, meeting); // → true
```

An event that starts on Sunday night and ends on Monday is only partly in the
week. `contains` says no; [`overlaps`](../../../api/minuta/overlaps/) says
they share time:

```ts
const party = period(
  Temporal.ZonedDateTime.from("2026-03-22T23:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-23T01:00[Europe/Zurich]")
);

contains(thisWeek, party); // → false
overlaps(thisWeek, party); // → true
```

## Do these bookings clash?

Two bookings clash when they share any time:

```ts
import { overlaps, period } from "@aleksejdix/minuta";

const at = (time) =>
  Temporal.ZonedDateTime.from(`2026-03-18T${time}[Europe/Zurich]`);

const first = period(at("09:00"), at("11:00"));
const second = period(at("10:30"), at("12:00"));
const third = period(at("11:00"), at("12:00"));

overlaps(first, second); // → true
overlaps(first, third); //  → false, third starts when first ends
```

The order does not matter: `overlaps(a, b)` is the same as `overlaps(b, a)`.
A period's `end` is not part of it, so back-to-back bookings like `first`
and `third` do not clash. Periods from `period`, `next` and `divide` never
overlap their neighbours either: each ends where the next begins.

## How many days are in this month?

[`count`](../../../api/minuta/count/) counts how many whole units fit
into a period:

```ts
import { count, month, year } from "@aleksejdix/minuta";

count(month(Temporal.PlainDate.from("2026-03-18")), "day"); // → 31
count(month(Temporal.PlainDate.from("2026-02-01")), "day"); // → 28
count(month(Temporal.PlainDate.from("2028-02-01")), "day"); // → 29, a leap year

count(year(Temporal.PlainDate.from("2026-01-01")), "day"); //  → 365
count(year(Temporal.PlainDate.from("2026-01-01")), "week"); // → 52
```

It counts on the calendar, so a day is not always 24 hours:

```ts
count(day(Temporal.PlainDate.from("2026-03-18")), "hour"); // → 24
count(day(Temporal.PlainDate.from("2026-03-29")), "hour"); // → 23, clocks go forward
```

Only **whole** units count. A 90-minute meeting is one hour, and March is four
weeks (31 days is not five):

```ts
count(
  period(
    Temporal.ZonedDateTime.from("2026-03-01T09:00[Europe/Zurich]"),
    Temporal.ZonedDateTime.from("2026-03-01T10:30[Europe/Zurich]")
  ),
  "hour"
); // → 1
count(month(Temporal.PlainDate.from("2026-03-18")), "week"); // → 4
```

Counting starts at the period's start, not at a unit border: from 14:00 one
day to 13:00 the next is 23 hours, so zero whole days.

## How long is this period?

Temporal gives the plain elapsed time: `until` from the start to the end, a
`Temporal.Duration`.

```ts
import { period } from "@aleksejdix/minuta";

const meeting = period(
  Temporal.ZonedDateTime.from("2026-03-18T09:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-18T11:00[Europe/Zurich]")
);

meeting.start.until(meeting.end).total("minutes"); // → 120

const night = period(
  Temporal.ZonedDateTime.from("2026-03-28T12:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-29T12:00[Europe/Zurich]")
);
night.start.until(night.end, { largestUnit: "hour" }).hours; // → 23, the clocks go forward
```

Use `until` for "how much time passes" (a timer, a progress bar). Use
`count` for "how many days or hours on the calendar" (a label like "31
days").

## The surprise: a period excludes its end

A period stops at the end you pass, and that value is not part of it. A
trip "from the 18th to the 25th" made like this leaves out the 25th:

```ts
import { contains, count, period } from "@aleksejdix/minuta";

const trip = period(
  Temporal.PlainDate.from("2026-03-18"),
  Temporal.PlainDate.from("2026-03-25")
);
// → { start: 2026-03-18, end: 2026-03-25, unit: "custom" }

contains(trip, Temporal.PlainDate.from("2026-03-25")); // → false
count(trip, "day"); // → 7
```

To include the whole last day, end the period where that day ends. The `end`
of a day period is the next date:

```ts
const wholeTrip = period(
  day(Temporal.PlainDate.from("2026-03-18")).start,
  day(Temporal.PlainDate.from("2026-03-25")).end
);
// → { start: 2026-03-18, end: 2026-03-26, unit: "custom" }

contains(wholeTrip, Temporal.PlainDate.from("2026-03-25")); // → true
count(wholeTrip, "day"); // → 8
```

  caption="trip stops where Mar 25 starts, so noon that day (the pin) is outside. wholeTrip ends with the day and takes it in."

:::tip
For a whole day, week or month, don't pass the edges yourself: use `day`,
`week` or `month`. They always end in the right place. Use `period(start, end)`
for spans that are not one whole unit, and end them at the first date or
moment that should be left out.
:::

`period` takes the start first. An end before the start throws a
`RangeError`: nothing is swapped behind your back.
