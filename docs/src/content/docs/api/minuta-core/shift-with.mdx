import { plainDate, zonedDateTime } from "../../../../components/temporal";
import {
  createUnits,
  periodWith,
  range,
  shiftWith,
} from "@aleksejdix/minuta/core";
import Timeline from "../../../../components/diagrams/Timeline.astro";

export const units = createUnits({ weekStartsOn: "sunday" });
export const week = periodWith(units, plainDate("2026-03-18"), "week");
export const slot = range(
  zonedDateTime("2026-03-02T09:00[Europe/Zurich]"),
  zonedDateTime("2026-03-02T10:00[Europe/Zurich]")
);

Move a period forward or back by a number of its own unit (two weeks ahead,
three months back), computed with the units you pass.

```ts
import { shiftWith } from "@aleksejdix/minuta/core";
```

```ts
shiftWith(units, period, steps);
```

## Example

```ts
import { createUnits, periodWith, shiftWith } from "@aleksejdix/minuta/core";

const units = createUnits({ weekStartsOn: "sunday" });
const week = periodWith(units, Temporal.PlainDate.from("2026-03-18"), "week");
// → { start: 2026-03-15, end: 2026-03-22, unit: "week" }

shiftWith(units, week, 2);
// → { start: 2026-03-29, end: 2026-04-05, unit: "week" }
```

## Parameters

| Name     | What it is                                                                          |
| -------- | ----------------------------------------------------------------------------------- |
| `units`  | The units to compute with, e.g. [`createUnits()`](../create-units/).                |
| `period` | The period to move.                                                                 |
| `steps`  | How many periods to move, as a whole number. Positive moves forward, negative back. |

## Returns

A new, whole period of the same unit, `steps` periods away. With `steps` of
`0`, the very same `period` object comes back.

It works exactly like [`shift`](../../minuta/shift/), which is `shiftWith`
with `createUnits()` bound. Read that page for how the period's own unit is
used and how custom periods move by their own length.

## Custom periods need no units

A period with the unit `"custom"` moves by its own length (days for a date
period, exact time for a time period), so `units` is not used for it at all. Any set of units, even an empty one,
gives the same result:

```ts
import { range, shiftWith } from "@aleksejdix/minuta/core";

const slot = range(
  Temporal.ZonedDateTime.from("2026-03-02T09:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-02T10:00[Europe/Zurich]")
);

shiftWith({}, slot, 2);
// → { start: 2026-03-02T11:00, end: 2026-03-02T12:00, unit: "custom" }
```

Pass the units the period was made with for every other period; see

## Errors

`shiftWith` throws a `RangeError` starting with `UNIT_NOT_SUPPORTED` when
`units` has no spec for `period.unit`.
