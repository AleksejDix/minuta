import { plainDate } from "../../../../components/temporal";
import { divide, period, range } from "@aleksejdix/minuta";
import { formatRange } from "@aleksejdix/minuta/format";
import Timeline from "../../../../components/diagrams/Timeline.astro";

export const ranges = [
  range(plainDate("2026-03-01"), plainDate("2026-04-01")),
  range(plainDate("2026-03-30"), plainDate("2026-04-06")),
  range(plainDate("2025-12-29"), plainDate("2026-01-05")),
];
export const q2 = period(plainDate("2026-05-01"), "quarter");
export const may = period(plainDate("2026-05-01"), "month");
export const may15 = period(plainDate("2026-05-15"), "day");

Show a period as "start – end" in any language. Parts that start and end
share, such as the month or the year, are written only once.

```ts
import { formatRange } from "@aleksejdix/minuta/format";
```

```ts
formatRange(period, locale, options?)
```

## Example

```ts
import { formatRange } from "@aleksejdix/minuta/format";
import { range } from "@aleksejdix/minuta";

const day = (iso) => Temporal.PlainDate.from(iso);

formatRange(range(day("2026-03-01"), day("2026-04-01")), "de-CH");
// → "1.–31. März 2026"

formatRange(range(day("2026-03-30"), day("2026-04-06")), "de-CH");
// → "30. März – 5. Apr. 2026"

formatRange(range(day("2025-12-29"), day("2026-01-05")), "de-CH");
// → "29. Dez. 2025 – 4. Jan. 2026"
```

The three ranges on one axis, each labelled with its `"de-CH"` text. What
start and end share is written once:

<Timeline
  caption="Inside one month, the month and the year appear once; across a month, only the year; across a year, nothing is shared."
  tracks={ranges.map((piece) => ({
    label: formatRange(piece, "de-CH"),
    periods: [piece],
    tone: "accent",
  }))}
/>

## Parameters

| Name      | What it is                                                                                 |
| --------- | ------------------------------------------------------------------------------------------ |
| `period`  | The period to show, of any unit.                                                           |
| `locale`  | The language and region, as a BCP 47 tag: `"en-US"`, `"de-CH"`, …                          |
| `options` | Optional. `Intl.DateTimeFormat` options merged over the defaults, see [Options](#options). |

## Returns

The range as a string, made by the built-in
`Intl.DateTimeFormat#formatRange`. A period of dates shows its first and its
last day: the end is excluded, so the label ends the day before `end`. A
period of moments shows its start and its end.

## What is shown depends on the unit

The period's unit picks which fields are shown, the same as for
[`formatPeriod`](../format-period/#one-format-per-unit). `"custom"`
periods show day, short month and year. So a custom range shows only dates,
no times:

```ts
const workday = range(
  Temporal.ZonedDateTime.from("2026-03-15T09:00[Europe/Zurich]"),
  Temporal.ZonedDateTime.from("2026-03-15T17:00[Europe/Zurich]")
);

formatRange(workday, "en-US");
// → "Mar 15, 2026" (9:00 and 17:00 are not shown)

formatRange(workday, "en-US", { hour: "numeric", minute: "2-digit" });
// → "9:00 AM – 5:00 PM"

formatRange(workday, "en-US", {
  day: "numeric",
  month: "short",
  year: "numeric",
  hour: "numeric",
  minute: "2-digit",
});
// → "Mar 15, 2026, 9:00 AM – 5:00 PM"
```

Options that pick fields replace the defaults, so list every field you want.

## Start and end can collapse to one

When start and end look the same in the chosen fields, there is no dash at
all:

```ts
import { period } from "@aleksejdix/minuta";

const day = (iso) => Temporal.PlainDate.from(iso);
const hour = Temporal.ZonedDateTime.from("2026-03-15T14:00[Europe/Zurich]");

formatRange(period(day("2026-03-01"), "month"), "de-CH"); // → "März 2026"
formatRange(period(day("2026-03-15"), "day"), "en-US"); // → "March 15, 2026"
formatRange(period(hour, "hour"), "en-US"); // → "Mar 15, 2:00 – 3:00 PM"
formatRange(period(day("2026-05-01"), "quarter"), "en-US"); // → "Apr – Jun 2026"
```

The same collapse on one axis: the quarter has a different first and last
month, so it gets a dash. May and May 15 start and end in the same month or
day, so they get none.

<Timeline
  caption="A quarter, one of its months and one of its days, each labelled with formatRange in en-US."
  tracks={[
    { label: formatRange(q2, "en-US"), periods: [q2], tone: "accent" },
    { label: formatRange(may, "en-US"), periods: [may], tone: "accent" },
    { label: formatRange(may15, "en-US"), periods: [may15], tone: "accent" },
    { label: "the days of May", periods: divide(may, "day"), tone: "muted" },
  ]}
/>

The hour shows its end as 3:00: a period of moments shows its `end`, where
the next hour starts. A period of dates shows its last day instead, so May
reads "May 2026", not "May 1 – Jun 1".

## Options

Options are those of `Intl.DateTimeFormat`, merged over the defaults for the
unit. Options that pick fields (`month`, `hour`, `dateStyle` …) replace the
defaults; others are added to them. Moments are written in their own zone;
pass `timeZone` to show them in another:

```ts
formatRange(period(hour, "hour"), "en-US", { timeZone: "America/New_York" });
// → "Mar 15, 9:00 – 10:00 AM"
```

See [`formatPeriod`](../format-period/#time-zones) for more.

## Errors

`formatRange` throws the errors of `Intl.DateTimeFormat`: a `RangeError` for
an invalid locale, a `TypeError` for conflicting options.
It does not check the order: a period whose end is before its start is
formatted as it is.
