import { plainDate, zonedDateTime } from "../../../../components/temporal";
import { divide, period, range } from "@aleksejdix/minuta";
import { isWeekendWith } from "@aleksejdix/minuta/calendar";
import { createUnits } from "@aleksejdix/minuta/core";
import MonthGrid from "../../../../components/diagrams/MonthGrid.astro";
import Timeline from "../../../../components/diagrams/Timeline.astro";

export const march = period(plainDate("2026-03-01"), "month");
export const days = divide(march, "day");
export const week = period(plainDate("2026-03-18"), "week");

Check whether a period lies completely on weekend days. The weekend is
Saturday and Sunday unless you set other days in the units.

```ts
import { isWeekendWith } from "@aleksejdix/minuta/calendar";
```

```ts
isWeekendWith(units, period);
```

## Example

```ts
import { isWeekendWith } from "@aleksejdix/minuta/calendar";
import { createUnits } from "@aleksejdix/minuta/core";
import { period } from "@aleksejdix/minuta";

const units = createUnits();

isWeekendWith(units, period(Temporal.PlainDate.from("2026-03-21"), "day")); // → true  (a Saturday)
isWeekendWith(units, period(Temporal.PlainDate.from("2026-03-18"), "day")); // → false (a Wednesday)
```

In a month view, the shaded days are the ones `isWeekendWith(units, day)`
returns `true` for: every Saturday and Sunday of March 2026, here with Sat
Mar 21 selected.

<MonthGrid month="2026-03" selected="2026-03-21" />

## Parameters

| Name     | What it is                                                                     |
| -------- | ------------------------------------------------------------------------------ |
| `units`  | The units to compute with, e.g. `createUnits()`. Only their `weekend` is used. |
| `period` | The period to check: a day, an hour, a range of days, …                        |

## Returns

`true` when every day the period touches is a weekend day, otherwise
`false`.

## Another weekend

Not every country rests on Saturday and Sunday. Set the weekend when you
create the units. Days are named in lowercase English; numbers work too
(0 = Sunday … 6 = Saturday).

```ts
const units = createUnits({ weekend: ["friday", "saturday"] });

isWeekendWith(units, period(Temporal.PlainDate.from("2026-03-20"), "day")); // → true  (a Friday)
isWeekendWith(units, period(Temporal.PlainDate.from("2026-03-22"), "day")); // → false (a Sunday)
```

The days of March 2026 for which `isWeekendWith` returns `true`, with the
default weekend and with Friday and Saturday. Every weekend moves one day
earlier:

<Timeline
  caption="Each bar is a day of March 2026 that counts as weekend with these units."
  tracks={[
    { label: "march", periods: [march], tone: "muted" },
    {
      label: '["saturday", "sunday"]',
      periods: days.filter((day) => isWeekendWith(createUnits(), day)),
      tone: "accent",
    },
    {
      label: '["friday", "saturday"]',
      periods: days.filter((day) =>
        isWeekendWith(createUnits({ weekend: ["friday", "saturday"] }), day)
      ),
      tone: "accent",
    },
  ]}
/>

With `weekend: []` there is no weekend, and `isWeekendWith` is always
`false`.

## Periods longer than a day

The check looks at every day the period touches, even partly. All of them
must be weekend days.

```ts
import { range } from "@aleksejdix/minuta";

const units = createUnits();

// Sat Mar 21 00:00 – Sun Mar 22 23:00
isWeekendWith(
  units,
  range(
    Temporal.ZonedDateTime.from("2026-03-21T00:00[Europe/Zurich]"),
    Temporal.ZonedDateTime.from("2026-03-22T23:00[Europe/Zurich]")
  )
);
// → true

// Fri Mar 20 18:00 – Sat Mar 21 10:00: Friday is not weekend
isWeekendWith(
  units,
  range(
    Temporal.ZonedDateTime.from("2026-03-20T18:00[Europe/Zurich]"),
    Temporal.ZonedDateTime.from("2026-03-21T10:00[Europe/Zurich]")
  )
);
// → false

isWeekendWith(units, period(Temporal.PlainDate.from("2026-03-18"), "week")); // → false
isWeekendWith(
  units,
  period(Temporal.ZonedDateTime.from("2026-03-21T10:00[Europe/Zurich]"), "hour")
); // → true (Saturday 10:00)
```

The week of Mar 16 2026 with its two weekend days on top. The first range
lies inside them; the second one starts on Friday evening and sticks out:

<Timeline
  caption="Only a period that stays inside the weekend days (top row) is weekend."
  tracks={[
    {
      label: "weekend days",
      periods: divide(week, "day").filter((day) =>
        isWeekendWith(createUnits(), day)
      ),
      tone: "muted",
    },
    {
      label: "Sat 00:00 – Sun 23:00: true",
      periods: [range(plainDate("2026-03-21"), plainDate("2026-03-23"))],
      tone: "accent",
    },
    {
      label: "Fri 18:00 – Sat 10:00: false",
      periods: [
        range(
          zonedDateTime("2026-03-20T18:00[Europe/Zurich]"),
          zonedDateTime("2026-03-21T10:00[Europe/Zurich]")
        ),
      ],
    },
    { label: "the week: false", periods: [week] },
  ]}
/>

A period that has both weekend days and working days is neither: both
`isWeekendWith` and [`isWeekdayWith`](../is-weekday-with/) return `false`
for it.

## Time zones

Which day it is depends on the time zone. A date period's days are its own
dates; a time period's days are read in its own zone. So the zone is decided
when the period is made. 22:00 UTC on Friday Mar 20 2026 is already Saturday
morning in Tokyo:

```ts
import { periodWith } from "@aleksejdix/minuta/core";

const tokyo = createUnits({ timeZone: "Asia/Tokyo" });
const moment = Temporal.Instant.from("2026-03-20T22:00:00Z");

isWeekendWith(tokyo, periodWith(tokyo, moment, "day")); // → true  (Saturday in Tokyo)
isWeekendWith(createUnits(), period(moment, "day")); // → false (Friday 23:00 in Zurich)
```

## Errors

`isWeekendWith` does not throw: it needs no unit specs, only the `weekend`
setting (Saturday and Sunday when the units have none). It does not check
the period: pass a valid one, e.g. from `period()`.
