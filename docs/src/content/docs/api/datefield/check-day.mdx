import { DateProblem, checkDay } from "@aleksejdix/datefield";
import MonthGrid from "../../../../components/diagrams/MonthGrid.astro";
import { plainDate } from "../../../../components/temporal";

export const rules = {
  isUnavailable: (day) => day.dayOfWeek >= 6,
  min: globalThis.Temporal.PlainDateTime.from("2026-06-15T18:30"),
};
export const monday = checkDay(plainDate("2026-06-15"), rules);
export const sunday = checkDay(plainDate("2026-06-14"), rules);
export const saturday = checkDay(plainDate("2026-06-20"), rules);
export const june = Array.from(
  { length: 30 },
  (_, index) => `2026-06-${String(index + 1).padStart(2, "0")}`
);
export const unavailable = june.filter(
  (day) => checkDay(plainDate(day), rules) === DateProblem.Unavailable
);

Check one day, a `Temporal.PlainDate`, against the same rules as
[`checkDate`](../check-date/), comparing whole days. Give it to a calendar's
`isDisabled`, and the calendar disables exactly the days the field reports.

```ts
import { checkDay } from "@aleksejdix/datefield";
```

```ts
checkDay(day, { min, max, isUnavailable });
```

## Example

```ts
import { checkDay } from "@aleksejdix/datefield";

const rules = {
  isUnavailable: (day) => day.dayOfWeek >= 6, // Saturday and Sunday
  min: Temporal.PlainDateTime.from("2026-06-15T18:30"),
};

checkDay(Temporal.PlainDate.from("2026-06-15"), rules); // → undefined
checkDay(Temporal.PlainDate.from("2026-06-14"), rules); // → "BEFORE_MIN"
checkDay(Temporal.PlainDate.from("2026-06-20"), rules); // → "UNAVAILABLE"
```

Here: Monday 15 June → <code>{String(monday)}</code>, Sunday 14 June → <code>{String(sunday)}</code>,
Saturday 20 June → <code>{String(saturday)}</code>. The `min` at 18:30 still allows all of
15 June.

<MonthGrid
  month="2026-06"
  from="2026-06-15"
  to="2026-06-30"
  marks={unavailable}
/>

June 2026 with these rules: the days from `min` on are highlighted, and the
weekends among them, which `checkDay` reports as `UNAVAILABLE`, have a dot. A
calendar given `isDisabled` disables the days before 15 June and the dotted ones.

## Parameters

| Name    | What it is                                                                  |
| ------- | --------------------------------------------------------------------------- |
| `day`   | The day to check, a `Temporal.PlainDate`, such as a calendar day's `start`. |
| `rules` | `min`, `max` and `isUnavailable`, as for [`checkDate`](../check-date/).     |

## Returns

A [`DateProblem`](../date-problem/) code, or `undefined` when the day is allowed.

## With the calendar

```tsx
import { checkDay } from "@aleksejdix/datefield";

<CalendarRoot isDisabled={(day) => checkDay(day.start, rules) !== undefined}>
  …
</CalendarRoot>;
```

The Vue `<CalendarRoot>` takes the same `isDisabled` prop.
