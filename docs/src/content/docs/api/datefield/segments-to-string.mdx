import { plainDate } from "../../../../components/temporal";
import {
  deriveFormat,
  fromTemporal,
  segmentsToString,
} from "@aleksejdix/datefield";
import FieldDiagram from "../../../../components/diagrams/FieldDiagram.astro";

export const written = (locale) =>
  segmentsToString(fromTemporal(plainDate("2026-03-01"), deriveFormat(locale)));

Join segments back into text: the values of all parts, one after another, so
the segments of `31.03.2026` give `"31.03.2026"` again.

```ts
import { segmentsToString } from "@aleksejdix/datefield";
```

```ts
segmentsToString(segments);
```

## Example

```ts
import {
  deriveFormat,
  fromTemporal,
  segmentsToString,
} from "@aleksejdix/datefield";

const format = deriveFormat("de-CH");

segmentsToString(fromTemporal(Temporal.PlainDate.from("2026-03-01"), format)); // → "01.03.2026"
segmentsToString(
  fromTemporal(Temporal.PlainDate.from("2026-03-01"), deriveFormat("en-US"))
); // → "03/01/2026"
segmentsToString(
  fromTemporal(Temporal.PlainDate.from("2026-03-01"), deriveFormat("ko-KR"))
); // → "2026. 03. 01."
```

The joined text of each, with the segments it was made of bracketed below.
The values are placed one after another; nothing is added between them.

## Parameters

| Name       | What it is                                                                                                  |
| ---------- | ----------------------------------------------------------------------------------------------------------- |
| `segments` | Segments, from [`parseSegments`](../parse-segments/), [`fromTemporal`](../from-temporal/) or after an edit. |

## Returns

One string: every segment's `value`, in order. Positions are not checked; the
values are simply joined.

## Show a date in a new field

Use it with `fromTemporal` to start a field with a value:

```ts
import {
  dateMask,
  deriveFormat,
  fromTemporal,
  segmentsToString,
} from "@aleksejdix/datefield";
import { createInputState } from "@aleksejdix/input-state";

const format = deriveFormat("de-CH");
const field = createInputState({
  mask: dateMask(format),
  value: segmentsToString(
    fromTemporal(Temporal.PlainDate.from("2026-03-31"), format)
  ),
});

field.buffer.text; // → "31.03.2026"
```

To put changed segments into an existing field, use
[`withSegments`](../with-segments/) instead. It calls `segmentsToString` and
keeps the cursor where it was.
