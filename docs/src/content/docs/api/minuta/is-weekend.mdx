import { plainDate, zonedDateTime } from "../../../../components/temporal";
import {
  createUnits,
  divide,
  isWeekend,
  month,
  period,
  week,
  withUnits,
} from "@aleksejdix/minuta";
import MonthGrid from "../../../../components/diagrams/MonthGrid.astro";

export const march = month(plainDate("2026-03-01"));
export const days = divide(march, "day");
export const thisWeek = week(plainDate("2026-03-18"));
export const fridayOff = withUnits(
  createUnits({ weekend: ["friday", "saturday"] })
);

Check whether a period lies completely on weekend days. The weekend is
Saturday and Sunday unless you set other days with `withUnits`.

```ts
import { isWeekend } from "@aleksejdix/minuta";
```

```ts
isWeekend(period);
```

## Example

```ts
import { day, isWeekend } from "@aleksejdix/minuta";

isWeekend(day(Temporal.PlainDate.from("2026-03-21"))); // → true  (a Saturday)
isWeekend(day(Temporal.PlainDate.from("2026-03-18"))); // → false (a Wednesday)
```

In a month view, the shaded days are the ones `isWeekend(day)`
returns `true` for: every Saturday and Sunday of March 2026, here with Sat
Mar 21 selected.

<MonthGrid month="2026-03" selected="2026-03-21" />

## Parameters

| Name     | What it is                                              |
| -------- | ------------------------------------------------------- |
| `period` | The period to check: a day, an hour, a range of days, … |

## Returns

`true` when every day the period touches is a weekend day, otherwise
`false`.

## Another weekend

Not every country rests on Saturday and Sunday. Set the weekend in
[`createUnits`](../create-units/) and take `isWeekend` from
[`withUnits`](../with-units/). Days are named in lowercase English; numbers
work too (0 = Sunday … 6 = Saturday). `createUnits({ locale: "ar-EG" })`
takes the weekend from a locale.

```ts
import { createUnits, day, withUnits } from "@aleksejdix/minuta";

const time = withUnits(createUnits({ weekend: ["friday", "saturday"] }));

time.isWeekend(day(Temporal.PlainDate.from("2026-03-20"))); // → true  (a Friday)
time.isWeekend(day(Temporal.PlainDate.from("2026-03-22"))); // → false (a Sunday)
```

The days of March 2026 for which `isWeekend` returns `true`, with the
default weekend and with Friday and Saturday. Every weekend moves one day
earlier:

  tracks={[

    {

With `weekend: []` there is no weekend, and `isWeekend` is always
`false`.

## Periods longer than a day

The check looks at every day the period touches, even partly. All of them
must be weekend days.

```ts
import { hour, isWeekend, period, week } from "@aleksejdix/minuta";

// Sat Mar 21 00:00 – Sun Mar 22 23:00
isWeekend(
  period(
    Temporal.ZonedDateTime.from("2026-03-21T00:00[Europe/Zurich]"),
    Temporal.ZonedDateTime.from("2026-03-22T23:00[Europe/Zurich]")
  )
);
// → true

// Fri Mar 20 18:00 – Sat Mar 21 10:00: Friday is not weekend
isWeekend(
  period(
    Temporal.ZonedDateTime.from("2026-03-20T18:00[Europe/Zurich]"),
    Temporal.ZonedDateTime.from("2026-03-21T10:00[Europe/Zurich]")
  )
);
// → false

isWeekend(week(Temporal.PlainDate.from("2026-03-18"))); // → false
isWeekend(hour(Temporal.ZonedDateTime.from("2026-03-21T10:00[Europe/Zurich]"))); // → true (Saturday 10:00)
```

The week of Mar 16 2026 with its two weekend days on top. The first range
lies inside them; the second one starts on Friday evening and sticks out:

<Timeline
  caption="Only a period that stays inside the weekend days (top row) is weekend."
  tracks={[
    {
      label: "weekend days",
      periods: divide(thisWeek, "day").filter((day) => isWeekend(day)),
      tone: "muted",
    },
    {
      label: "Sat 00:00 – Sun 23:00: true",

        ),
      ],
    },
    { label: "the week: false", periods: [thisWeek] },
  ]}
/>

A period that has both weekend days and working days is neither: both
`isWeekend` and [`isWeekday`](../is-weekday/) return `false`
for it.

## Time zones

Which day it is depends on the time zone. A date period's days are its own
dates; a time period's days are read in its own zone. So the zone is decided
when the period is made. 22:00 UTC on Friday Mar 20 2026 is already Saturday
morning in Tokyo:

```ts
import { day, isWeekend } from "@aleksejdix/minuta";

const moment = Temporal.Instant.from("2026-03-20T22:00:00Z");

isWeekend(day(moment.toZonedDateTimeISO("Asia/Tokyo"))); // → true  (Saturday in Tokyo)
isWeekend(day(moment.toZonedDateTimeISO("Europe/Zurich"))); // → false (Friday 23:00 in Zurich)
```

## Errors

`isWeekend` does not throw: it needs no unit specs, only the `weekend`
setting (Saturday and Sunday when the units have none). It does not check
the period: pass a valid one, e.g. from `period()`.
