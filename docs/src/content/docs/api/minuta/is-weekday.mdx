import { plainDate } from "../../../../components/temporal";
import {
  createUnits,
  divide,
  isWeekday,
  month,
  period,
  week,
  withUnits,
} from "@aleksejdix/minuta";
import MonthGrid from "../../../../components/diagrams/MonthGrid.astro";
import Timeline from "../../../../components/diagrams/Timeline.astro";

export const march = month(plainDate("2026-03-01"));
export const days = divide(march, "day");
export const thisWeek = week(plainDate("2026-03-18"));
export const iso = (piece) => piece.start.toString();
export const fridayOff = withUnits(
  createUnits({ weekend: ["friday", "saturday"] })
);

Check whether a period lies completely on working days, that is, days that
are not part of the weekend. The weekend is Saturday and Sunday unless you
set other days with `withUnits`.

```ts
import { isWeekday } from "@aleksejdix/minuta";
```

```ts
isWeekday(period);
```

## Example

```ts
import { day, divide, isWeekday, month } from "@aleksejdix/minuta";

isWeekday(day(Temporal.PlainDate.from("2026-03-18"))); // → true  (a Wednesday)
isWeekday(day(Temporal.PlainDate.from("2026-03-21"))); // → false (a Saturday)

// The working days of March 2026
divide(month(Temporal.PlainDate.from("2026-03-01")), "day").filter((piece) =>
  isWeekday(piece)
).length;
// → 22
```

The dots mark those 22 days: every day of March 2026 for which
`isWeekday(day)` returns `true`. The shaded weekend days have
none.

<MonthGrid
  month="2026-03"
  marks={days.filter((day) => isWeekday(day)).map(iso)}
/>

## Parameters

| Name     | What it is                                              |
| -------- | ------------------------------------------------------- |
| `period` | The period to check: a day, an hour, a range of days, … |

## Returns

`true` when no day the period touches is a weekend day, otherwise `false`.

## Another weekend

Set the weekend in [`createUnits`](../create-units/), by hand or from a
locale (`createUnits({ locale: "ar-EG" })`), and take `isWeekday` from
[`withUnits`](../with-units/). Then Sunday is a working day:

```ts
import { createUnits, day, withUnits } from "@aleksejdix/minuta";

const time = withUnits(createUnits({ weekend: ["friday", "saturday"] }));

time.isWeekday(day(Temporal.PlainDate.from("2026-03-22"))); // → true  (a Sunday)
time.isWeekday(day(Temporal.PlainDate.from("2026-03-20"))); // → false (a Friday)
```

The working days of March 2026 with both settings. With Friday and Saturday
off, Sundays become working days, and March has 23 of them:

<Timeline

  tracks={[
    { label: "march", periods: [march], tone: "muted" },
    {
      label: '["saturday", "sunday"]',

minuta knows nothing about public holidays. A holiday on a Tuesday is a
working day for `isWeekday`.

## Periods longer than a day

Every day the period touches counts, even partly:

```ts
import { isWeekday, month, period, week } from "@aleksejdix/minuta";

// Mon Mar 16 – Fri Mar 20 (the end, Sat Mar 21, is excluded)
isWeekday(
  period(
    Temporal.PlainDate.from("2026-03-16"),
    Temporal.PlainDate.from("2026-03-21")
  )
);
// → true

isWeekday(week(Temporal.PlainDate.from("2026-03-18"))); // → false (has Sat and Sun)
isWeekday(month(Temporal.PlainDate.from("2026-03-01"))); // → false
```

      tone: "accent",

A period with both kinds of days is neither: both `isWeekday` and
[`isWeekend`](../is-weekend/) return `false`.

A time period's days are read in its own zone, see
[`isWeekend`](../is-weekend/#time-zones).

## Errors

`isWeekday` does not throw: it needs no unit specs, only the `weekend`
setting (Saturday and Sunday when the units have none). It does not check
the period: pass a valid one, e.g. from `period()`.
